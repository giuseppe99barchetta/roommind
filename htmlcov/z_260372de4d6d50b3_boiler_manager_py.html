16class BoilerState(StrEnum):

18 PREOPENING = "preopening" 

19 ON = "on"

24class BoilerManager: 

25 """Enforce: an enabled boiler always has a verified hydraulic path. 

26 

27 Delays are state timestamps rather than sleeps, keeping the HA event loop 

28 responsive and making restart recovery conservative. 

29 """ 

30 

31 def __init__(self, hass: HomeAssistant) -> None: 

32 self.hass = hass 

33 self.state = BoilerState.OFF 

34 self._state_since = monotonic() 

35 self.demand_rooms: set[str] = set() 

36 self.path_safe = False

38 @staticmethod 

39 def _available(hass: HomeAssistant, entity_id: str) -> bool: 

40 st = hass.states.get(entity_id) 

41 return bool(st and st.state not in ("unknown", "unavailable")) 

42 

43 async def _set_bypass(self, settings: dict, open_: bool) -> bool: 

44 bypasses = [e for e in settings.get("hydraulic_bypass_entities", []) if e] 

45 if not bypasses: 

46 return False 

47 if not open_: 

48 # Releasing control is intentionally conservative: RoomMind idles 

49 # the bypasses only after the boiler has been stopped and held. 

50 for entity_id in bypasses: 

51 try: 

52 await self.hass.services.async_call( 

53 "climate", 

54 "set_hvac_mode", 

55 {"entity_id": entity_id, "hvac_mode": "off"}, 

56 blocking=True, 

57 context=make_roommind_context(), 

58 ) 

59 except Exception: # noqa: BLE001 

60 _LOGGER.warning("Unable to release boiler bypass '%s'", entity_id, exc_info=True) 

61 return True 

62 target = float(settings.get("hydraulic_bypass_open_temperature", 28.0)) 

63 verified = False 

64 for entity_id in bypasses: 

65 if not self._available(self.hass, entity_id): 

66 continue 

67 try: 

68 await self.hass.services.async_call( 

69 "climate", 

70 "set_hvac_mode", 

71 {"entity_id": entity_id, "hvac_mode": "heat"}, 

72 blocking=True, 

73 context=make_roommind_context(), 

74 ) 

75 await self.hass.services.async_call( 

76 "climate", 

77 "set_temperature", 

78 {"entity_id": entity_id, "temperature": target}, 

79 blocking=True, 

80 context=make_roommind_context(), 

81 ) 

82 verified = True 

83 except Exception: # noqa: BLE001 

84 _LOGGER.warning("Unable to open boiler bypass '%s'", entity_id, exc_info=True) 

85 return verified

87 async def _set_boiler(self, settings: dict, on: bool) -> bool: 

88 entity_id = settings.get("boiler_entity", "") 

89 if not self._available(self.hass, entity_id): 

90 return False 

91 control_type = settings.get("boiler_control_type", "climate") 

92 domain = "switch" if control_type == "switch" else "climate" 

93 service = ( 

94 "turn_on" if on and domain == "switch" else "turn_off" if not on and domain == "switch" else "set_hvac_mode" 

95 ) 

96 data = {"entity_id": entity_id} 

97 if service == "set_hvac_mode": 

98 data["hvac_mode"] = "heat" if on else "off" 

99 try: 

100 await self.hass.services.async_call(domain, service, data, blocking=True, context=make_roommind_context()) 

101 return True 

102 except Exception: # noqa: BLE001 

103 _LOGGER.warning("Unable to change boiler '%s'", entity_id, exc_info=True) 

104 return False

106 async def async_reconcile(self, settings: dict, demand_rooms: set[str]) -> None: 

107 """Advance one safe transition. Missing safety configuration disables boiler.""" 

108 if not settings.get("climate_control_active", True): 

109 demand_rooms = set() 

110 self.demand_rooms = set(demand_rooms) 

111 if not settings.get("boiler_entity"): 

112 self.state = BoilerState.OFF 

113 self.path_safe = False 

114 return 

115 now = monotonic() 

116 demand = bool(demand_rooms) 

117 pre_delay = float(settings.get("boiler_startup_delay_seconds", 30)) 

118 post_delay = float(settings.get("boiler_shutdown_delay_seconds", 60)) 

119 if self.state == BoilerState.OFF and demand: 

120 self.path_safe = await self._set_bypass(settings, True) 

121 if not self.path_safe: 

122 self.state = BoilerState.FAULT 

123 await self._set_boiler(settings, False) 

124 return 

125 self.state, self._state_since = BoilerState.PREOPENING, now 

126 _LOGGER.info("Boiler demand %d: opening hydraulic bypass", len(demand_rooms)) 

127 elif self.state == BoilerState.PREOPENING: 

128 self.path_safe = await self._set_bypass(settings, True) 

129 if not demand: 

130 await self._set_boiler(settings, False) 

131 await self._set_bypass(settings, False) 

132 self.path_safe = False 

133 self.state, self._state_since = BoilerState.OFF, now 

134 elif not self.path_safe: 

135 self.state = BoilerState.FAULT 

136 await self._set_boiler(settings, False) 

137 elif now - self._state_since >= pre_delay: 

138 if await self._set_boiler(settings, True): 

139 self.state, self._state_since = BoilerState.ON, now 

140 _LOGGER.info("Boiler hydraulic path verified; starting boiler") 

141 else: 

142 self.state = BoilerState.FAULT 

143 elif self.state == BoilerState.ON: 

144 self.path_safe = await self._set_bypass(settings, True) 

145 if not self.path_safe: 

146 _LOGGER.error("Hydraulic path lost while boiler active; shutting boiler down") 

147 await self._set_boiler(settings, False) 

148 self.state, self._state_since = BoilerState.FAULT, now 

149 elif not demand: 

150 await self._set_boiler(settings, False) 

151 self.state, self._state_since = BoilerState.POSTSTOP, now 

152 elif self.state == BoilerState.POSTSTOP: 

153 if demand: 

154 self.path_safe = await self._set_bypass(settings, True) 

155 self.state, self._state_since = BoilerState.PREOPENING, now 

156 elif now - self._state_since >= post_delay: 

157 await self._set_bypass(settings, False) 

158 self.path_safe = False 

159 self.state, self._state_since = BoilerState.OFF, now 

160 elif self.state == BoilerState.FAULT: 

161 await self._set_boiler(settings, False) 

162 if not demand: 

163 self.path_safe = False 

164 self.state, self._state_since = BoilerState.OFF, now
