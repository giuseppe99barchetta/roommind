1"""Rooms whose Home Assistant area was deleted rest (#494). 

2 

3A resting room keeps its configuration, learned model and history but has no entities and 

4drives no device; it runs again once its area is back. "Resting" is not stored: it is "stored, 

5but its area is missing from the area registry". ``rest_switched_off`` marks a resting room whose 

6shutdown is complete (every device it drove has an outcome, see ``PendingShutdowns``). 

7""" 

8 

9from __future__ import annotations 

10 

11from homeassistant.core import HomeAssistant 

12from homeassistant.helpers import area_registry as ar 

13 

14REST_SWITCHED_OFF = "rest_switched_off" 

15REST_SHUTDOWN_EXPIRED = "rest_shutdown_expired" 

16

18def has_area(hass: HomeAssistant, area_id: str) -> bool: 

19 """True while *area_id* exists in the area registry (always True without a loaded registry).""" 

20 registry = hass.data.get(ar.DATA_REGISTRY) if isinstance(getattr(hass, "data", None), dict) else None 

21 # Not loaded yet (no `areas` before async_load): every room runs 

22 areas = getattr(registry, "areas", None) if registry is not None else None 

23 if areas is None: 

24 return True 

25 return area_id in areas 

26 

27 

28def rooms_with_area(hass: HomeAssistant, rooms: dict[str, dict]) -> dict[str, dict]: 

29 """The running rooms of *rooms*.""" 

30 return {area_id: room for area_id, room in rooms.items() if has_area(hass, area_id)} 

31 

32 

33def rooms_without_area(hass: HomeAssistant, rooms: dict[str, dict]) -> dict[str, dict]: 

34 """The resting rooms of *rooms*.""" 

35 return {area_id: room for area_id, room in rooms.items() if not has_area(hass, area_id)}
